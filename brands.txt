// BRANDS BAŞLANGIÇ

$sql = "SELECT * FROM brands";
$result = $conn->query($sql);

$brands = [];

if ($result->num_rows > 0) {
    while ($row = $result->fetch_assoc()) {
        $brandID = $row['brandID'];
        $brandName = $row['brandName_tr'];
        $imgID = $row['imgID'];

        $imgQuery = "SELECT imgURL FROM images WHERE imgID = $imgID";
        $imgResult = $conn->query($imgQuery);

        if ($imgResult->num_rows > 0) {
            $imgRow = $imgResult->fetch_assoc();
            $brandImgURL = $imgRow['imgURL'];
        } else {
            $brandImgURL = "";
            echo "Marka resmi bulunamadı.";
        }

        $brands[] = [
            'brandName' => $brandName,
            'imgURL' => $brandImgURL
        ];
    }
} else {
    echo "Marka verisi bulunamadı.";
}
